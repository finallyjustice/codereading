QEMU RunState Transition Examples.

RUN_STATE_PRELAUNCH to RUN_STATE_SUSPENDED.

QEMU-10.2.0.

qemu-system-x86_64 \
-machine q35,dump-guest-core=off \
-accel kvm -cpu host \
-name debug-threads=on \
-m 8G -smp 4 \
-vnc :8 -monitor stdio -S \
-device virtio-scsi-pci,id=scsi0,num_queues=4 \
-no-shutdown \
-hda ol89.qcow2

(qemu) info status
VM status: paused (prelaunch)

(qemu) cont

[root@vm ~] echo mem > /sys/power/state

(qemu) info status
VM status: paused (suspended)

(qemu) savevm 0001

# Create QEMU instance again.

(qemu) info status
VM status: paused (prelaunch)

(qemu) loadvm 0001

(qemu) info status
VM status: paused (prelaunch)

(qemu) cont

(qemu) info status
VM status: paused (suspended)
