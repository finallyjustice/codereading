

virtio scsi probe的过程.
假设hotplug一个vhost-scsi的HBA, 上面只有一个lun.

virtscsi_probe()
-> scsi_scan_host()
   -> async_schedule(do_scan_async, data)


do_scan_async()
-> do_scsi_scan_host()
   -> scsi_scan_host_selected(shost, SCAN_WILD_CARD, SCAN_WILD_CARD, SCAN_WILD_CARD, SCSI_SCAN_INITIAL);
      -> 针对每一个channel: scsi_scan_channel() 一般一个channel
         -> 针对256个target: __scsi_scan_target()


__scsi_scan_target() -->调用256次
-> scsi_alloc_target()
   -> scsi_probe_and_add_lun() -->针对LUN0, 返回SCSI_SCAN_LUN_PRESENT或者SCSI_SCAN_TARGET_PRESENT说明有效
      -> scsi_alloc_sdev()
      -> __scsi_remove_device()
   -> scsi_report_lun_scan()
      if (!scsi_device_lookup_by_target(starget, 0))
        scsi_alloc_sdev(LUN0) ===> 注意, 就算LUN0不存在这里重新分配!
      -> 针对每一个lun调用scsi_probe_and_add_lun()
         -> scsi_alloc_sdev()
         -> __scsi_remove_device()
